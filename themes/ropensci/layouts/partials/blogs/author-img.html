{{- /* Returns the URL of an author's picture: their "img" if set, else their GitHub avatar if "github" is set, else an identicon generated from their name. Context: the author's page. */ -}} {{- $result := "" -}} {{- if isset .Params "img" -}} {{- $result = print "/" .Params.img | absURL -}} {{- else if isset .Params "github" -}} {{- $result = print "https://github.com/" .Params.github ".png" -}} {{- else -}} {{- $result = print "https://api.dicebear.com/7.x/identicon/svg?seed=" (md5 .Params.name) "&rowColor=276dcc&size=50" -}} {{- end -}} {{- return $result -}}
