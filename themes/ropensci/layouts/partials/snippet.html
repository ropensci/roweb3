{{- /* Returns a Markdown snippet from assets/_snippets/, rendered to HTML. Context: a dict with "name" (e.g. "coworking") and "lang" (the page's language, e.g. "en"). Fails the build when assets/_snippets/..md does not exist. */ -}} {{- $snippetPath := printf "_snippets/%s.%s.md" .name .lang -}} {{- with resources.Get $snippetPath -}} {{- return (.Content | markdownify) -}} {{- else -}} {{- errorf "Snippet %q not found for language %q: add the file themes/ropensci/assets/%s." .name .lang $snippetPath -}} {{- end -}}
