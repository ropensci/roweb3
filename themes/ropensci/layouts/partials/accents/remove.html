{{- /* Returns a string with its accented letters replaced by unaccented ones (e.g. "é" by "e"), using the replacements listed in data/accents/accents.json. Context: the string. */ -}} {{- $result := . -}} {{- range $replacement := hugo.Data.accents.accents -}} {{- range $accented, $unaccented := $replacement -}} {{- $result = replaceRE $accented $unaccented $result -}} {{- end -}} {{- end -}} {{- return $result -}}
