{{- /* "Our reviewers" section of the software-review page: each reviewer links to their GitHub profile, followed by links to their reviews, in a list collapsed by default. Context: none, reads data/reviewers/reviewers.json (written by scripts/update-reviewers.R from Airtable). Fails the build if the file is missing, or if a reviewer lacks a github, name or reviews entry, or a review lacks a url or date. */ -}} {{- $reviewers := hugo.Data.reviewers.reviewers -}} {{- $howToFix := "Run scripts/update-reviewers.R to regenerate data/reviewers/reviewers.json." -}} {{- if not $reviewers -}} {{- errorf "data/reviewers/reviewers.json is missing or empty. %s" $howToFix -}} {{- end -}} {{- range $reviewer := $reviewers -}} {{- if not (and $reviewer.github $reviewer.name $reviewer.reviews) -}} {{- errorf "Reviewer %q (GitHub %q) in data/reviewers/reviewers.json lacks a github, name or reviews entry. %s" $reviewer.name $reviewer.github $howToFix -}} {{- end -}} {{- range $review := $reviewer.reviews -}} {{- if not (and $review.url $review.date) -}} {{- errorf "A review by %q in data/reviewers/reviewers.json lacks a url or date (url %q, date %q). %s" $reviewer.github $review.url $review.date $howToFix -}} {{- end -}} {{- end -}} {{- end }}
